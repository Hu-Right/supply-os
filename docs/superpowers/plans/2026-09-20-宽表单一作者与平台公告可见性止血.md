```ts
/**
 * 搜索可见性清除出口
 * Search Visibility Purge Outlet
 *
 * @module server/services/search-visibility/purge
 * @description 「让某公告对公众不可见」的唯一清除出口：删宽表行 + 联动删 Meilisearch
 *              文档 + 失效搜索结果缓存。平台公告状态转非发布、外部后台删除公告等场景
 *              必须走本出口，禁止各处再各写一份 DELETE（口径分叉是宽表震荡缺陷的根因）。
 *
 *              与 I1（宽表单一写入者）的关系：本模块只做**删除**，不写任何业务列内容；
 *              内容写入的唯一路径仍是 buildWideRow → upsertWideRows。
 *
 *              级联语义复用 syncNoticeIds：宽表行已删 + 主表行不可公开可见 ⇒ 删索引文档
 *              （见 meilisearch/sync.ts 的 ghost 判定），不在此处直连 Meili 客户端，
 *              避免第二套「删除文档」实现与重试/降级路径分叉。
 */
import type { Pool } from "mysql2/promise";
import { syncNoticeIds } from "../meilisearch";
import { invalidateSearchCache } from "../search-common/sync-events";

/** 单批 IN 上限：与 sync-scheduler 的增量批同量级，避免超长 IN 语句 */
const PURGE_BATCH = 500;

/**
 * 从搜索侧彻底移除指定公告（宽表行 + 索引文档 + 结果缓存）。
 * @returns 宽表实际删除行数
 */
export async function purgeNoticeSearch(pool: Pool, ids: number[]): Promise<number> {
  const clean = [...new Set(ids.filter((id) => Number.isFinite(id) && id > 0))];
  if (clean.length === 0) return 0;

  let deleted = 0;
  for (let i = 0; i < clean.length; i += PURGE_BATCH) {
    const batch = clean.slice(i, i + PURGE_BATCH);
    const placeholders = batch.map(() => "?").join(",");
    const [res] = await pool.query(
      `DELETE FROM crm_notice_search WHERE id IN (${placeholders})`,
      batch,
    );
    deleted += Number((res as { affectedRows?: number }).affectedRows ?? 0);
    // 级联删索引文档：宽表已无行 + 主表不可见 → syncNoticeIds 的 ghost 判定命中删除分支
    await syncNoticeIds(pool, batch).catch((err: unknown) => {
      console.warn("[search-purge] Meilisearch 级联删除失败（交由定时对账兜底）:", (err as Error).message);
    });
  }
  invalidateSearchCache();
  return deleted;
}
```

```ts
 * 平台公告状态漂移检测（D2 修复，只检测不修复 —— I1）
 *
 * 背景：宽表增量同步是 `id > watermark` 的纯新行扫描，而审核通过由站外后台直接写库
 *      （红线 5 允许的「受控后台直接维护数据」），老行 rfq_status 变 published 后
 *      其 id 早已在水位之下，永远不会被增量拉取。
 *
 * 口径：可见性 = `entry_source='platform'` 且 `rfq_status='published'`，与
 *      lib/utils/notice-expired 的 PLATFORM_PUBLISHED_ONLY 完全一致；
 *      过期不改变可见性（过期行由 deadline_sec 在查询侧过滤），故不纳入本检测。
 *
 * 成本：走 idx_entry_source（迁移 069），平台行数量为「用户发布量」级（远小于爬虫存量）。
 */
export async function detectPlatformStatusDrift(pool: Pool): Promise<PlatformDrift> {
  const [rows] = await pool.query(
    `SELECT n.id, ns.id AS wide_id
     FROM crm_bid_notices n
     LEFT JOIN crm_notice_search ns ON ns.id = n.id
     WHERE n.entry_source = 'platform'
       AND (
         (IFNULL(n.rfq_status, '') = ${JSON.stringify(RFQ_STATUS.PUBLISHED)} AND ns.id IS NULL)
         OR (IFNULL(n.rfq_status, '') <> ${JSON.stringify(RFQ_STATUS.PUBLISHED)} AND ns.id IS NOT NULL)
       )
     LIMIT 500`,
  );
  const toSync: number[] = [];
  const toPurge: number[] = [];
  for (const r of rows as RowDataPacket[]) {
    if (r.wide_id == null) toSync.push(Number(r.id));
    else toPurge.push(Number(r.id));
  }
  return { toSync, toPurge };
}
```

```ts
 * 宽表源指纹（Single-Writer 的收敛基础）
 * Wide Table Source Fingerprint
 *
 * @module server/services/search-sync/wide-fingerprint
 * @description 宽表历史上存在两个作者：buildWideRow（构建）与 5 个手写 UPDATE（对账修复），
 *              两者口径分叉导致「每轮判漂移 → 改错 → 下轮又改回」的震荡（D4/D5/D7）与
 *              永不收敛。本模块把「宽表内容是否落后于输入」收敛为一个可比较的指纹：
 *

 *                 修复一律委托 syncWideIds（即重建），故重建后必然相等（可断言收敛）。
 *              3. 指纹只依赖输入、绝不读宽表自身列，避免自反馈。
 *

 *              group_concat_max_len（默认 1024 字节）截断，超限后的内容变化指纹看不见。

 * 全库唯一定义点：WIDE_SYNC_SELECT 与 reconcileFingerprint 都拼本常量，禁止再写第二份。

  IFNULL(LEFT(n.reference, ${WIDE_LIMITS.reference}), ''),

          FROM crm_notice_translations WHERE notice_id = n.id), 'x'),

          FROM crm_bid_opportunities o2
          JOIN crm_bid_opportunity_unspsc_candidates c
            ON c.opportunity_id = o2.id AND c.status = 'approved'
          WHERE o2.source_notice_id = n.notice_id), 'x')

/**
 * 检测「宽表内容落后于输入」的公告 id（只读，不修复 —— I1）。
 * @returns 指纹不等的 id 列表；交由 syncWideIds 重建
 */
export async function detectWideFingerprintDrift(pool: Pool): Promise<number[]> {
  const ids: number[] = [];
  for (let i = 0; i < FP_SLICES_PER_ROUND; i++) {
    const [rows] = await pool.query(
      `SELECT n.id, ${WIDE_FP_EXPR} AS fp, ns.sync_src_hash
       FROM crm_bid_notices n
       ${WIDE_OPP_JOIN}
       INNER JOIN crm_notice_search ns ON ns.id = n.id
       WHERE ns.id > ?
       ORDER BY ns.id ASC
       LIMIT ${FP_SLICE_SIZE}`,
      [_fpCursor],
    );
    const batch = rows as Array<{ id: number; fp: string; sync_src_hash: string }>;
    if (batch.length === 0) {
      // 已到表尾：归零游标，下一轮从头开始（全表周期扫描）
      _fpCursor = 0;
      break;
    }
    _fpCursor = Number(batch[batch.length - 1].id);
    for (const r of batch) {
      if (String(r.sync_src_hash ?? "") !== String(r.fp ?? "")) ids.push(Number(r.id));
    }
    if (batch.length < FP_SLICE_SIZE) {
      _fpCursor = 0;
      break;
    }
  }
  return ids;
}
```

```ts
function mergeIdLists(...lists: Array<string | number | null | undefined>): string {
  const out: string[] = [];
  for (const l of lists) {

      if (v && !out.includes(v)) out.push(v);
```

```js
const LIB_DIR = path.join(ROOT, "src", "lib");

    return;
```

```markdown
## 结论

`crm_notice_search` 的内容（含一致性指纹 `sync_src_hash`）只由 `buildWideRow → upsertWideRows` 写入；对账层退化为「只检测、返回待重建 id」；跨层口径抽为常量单一出口；并新增 `check-wide-table-ssot.mjs` 门禁把该标准变成机器强制。

## 理由与权衡

宽表历史上同时被「构建路径」与 5 段手写 `UPDATE` 修复路径写入，两份口径各自演进必然分叉，已实测产生：精选公告每 5 分钟震荡（描述来源不一致）、`precise_levelN` 被写入 UNSPSC 原码（列语义污染 + 每轮误判）、精选人工摘要被机器译文覆盖、同语言直通判定值写错导致描述恒空、内容漂移检测因批次算法在生产几乎不生效。逐处对齐 SQL 能止住当下症状，但 bug 类会随任一侧改动复发。指纹方案把「是否落后」变成一次可比较的哈希，从结构上取消第二个作者，且收敛性可被自动断言。

## 被否决的替代方案

- **逐处修正 5 段对账 UPDATE 的表达式**：改动最小、当天可上，但口径仍存于两处，属"下一次必然复发"的修法。
- **Outbox / CDC 事件表彻底摆脱 id 水位**：最干净，但要求站外审核后台改为写事件表；该后台不在本仓库，跨系统协调成本与本轮收益不匹配。若将来出现「对账延迟造成业务事故」的真实案例，由 `reconcilePlatformStatus` 升级为 outbox 消费取代。
- **把 `rfq_status` 冗余进宽表并在查询侧过滤**：可避免发布/撤回的删增震荡，但代价是「未发布内容进入索引与 SSR 面」的合规风险，以及每个消费方都要带该过滤条件。

## 生效与失效条件

生效：`src/lib/**` 无 `UPDATE crm_notice_search`（门禁保证）；连续两轮全量对账变更数为 0（测试断言）。
失效：若 `crm_agency_aliases` / UNSPSC 字典树变更为高频事件，则"不进指纹"的边界不再成立，需将维度版本纳入指纹或改为定时全量重建；若站外后台可控改造，则本方案的对账兜底应被 Outbox 取代。
```
